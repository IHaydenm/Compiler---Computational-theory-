int main(){
    int temperature = 30;
    bool is_hot = false;
    bool is_cold = false;

    if (temperature > 25) {
        is_hot = true;
        // Block statement within if
        int status = 1;
    }
    elif (temperature < 10) {
        is_cold = true;
    }
    else {
        // Between 10 and 25 (inclusive)
        is_hot = false;
        is_cold = false;
    }

    int check_val = 5;
    
    if (check_val == 5) {
        int result = 1;
    } else {
        int result = 0;
    }

    // Nested conditional
    if (is_hot) {
        if (temperature > 40) {
            int too_hot = 1;
        }
    }

    // Another conditional check
    int humidity = 70;
    if (temperature > 20 && humidity > 60) {
        int muggy = 1;
    }

    // Comparing mixed types (int and float)
    float limit = 29.5;
    if (temperature > limit) {
        int exceeded = 1;
    }

    // Complex boolean condition
    bool sunny = true;
    if (is_hot && sunny) {
        int complex_decision = 1;
    }
}
