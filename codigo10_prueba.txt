int main() {
    int a = 1;
    float b = 2.5;

    // Use of undeclared variable (will cause semantic error)
    // int result = a + undeclared_var; // Semantic error: use of undeclared variable

    // Assignment to undeclared variable (will cause semantic error)
    // another_var = a + b; // Semantic error: assignment to undeclared variable

    float c;

    // Correct declarations
    int result = a + 5;
    
    // Correct assignment
    c = a + b;

    // Correct loop
    for (int i = 0; i < 5; i = i + 1) {
        c = c + i;
    }

    // Correct conditional
    if (a < 10) {
        c = c + 1;
    }
}
