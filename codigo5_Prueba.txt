int main() {
    int i = 5;
    float f = 10.5;
    bool b = true;

    // int + float -> float
    float result1 = i + f;

    // float * int -> float
    float result2 = f * i;

    // int = float (allowed due to implicit float to int conversion not checked)
    // The semantic checker allows float assignment to int, but prints an error
    // int check_assign = f; // Will cause semantic error: cannot assign float to int
    
    // float = int (allowed due to int to float promotion)
    float promotion = i;

    // Relational operators on mixed types -> bool
    bool comparison1 = i < f;
    bool comparison2 = f >= 5;

    // Relational operator on bool (semantic error will occur)
    // bool error_comp = b > true; // Should cause semantic error

    // Arithmetic on bool (semantic error will occur)
    // int error_arith = b + 5; // Should cause semantic error
    
    // Logical operators only accept bool types
    bool logic_op;
    
    // Unary NOT only accepts bool types
    bool not_b;

    // Nested expression with mixed types
    float final_calc = (i + 2) * (f / 2.0);
    
    // A correct comparison recognized by the lexer:
    bool less_equal = i <= f;
}
