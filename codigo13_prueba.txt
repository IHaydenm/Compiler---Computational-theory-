int add(int a, int b) {<-- NOT SUPPORTED (parameters)
    int result = a + b;            
    return result;                 
}                                  

int main() {                       
    int x = 5;                     
    int y = 7;                                                        
    int z = add(x, y);                                                
    if (z > 10) {                  
        z = z + 1;                 
    } else {                       
        z = z - 1;                 
    }                              
    for (int i = 0; i < z; i = i + 1) { 
        x = x + i;                 
    }                              
                                   
    int c = 0;                     
    while (c < 3) {                
        c = c + 1;                 
    }                              
                                   
    return z;                      
}
